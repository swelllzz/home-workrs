import random

# Компьютер загадывает случайное число от 1 до 100
secret = random.randint(1, 100)

# Количество доступных попыток
attempts = 7

print("Игра «Компьютер и пользователь»")
print("Компьютер загадал число от 1 до 100. У вас есть 7 попыток, чтобы его угадать!")

# Цикл выполняется, пока есть попытки
while attempts > 0:
    print(f"\nОсталось попыток: {attempts}")
    
    # Получаем число от пользователя
    try:
        user_guess = int(input("Введите ваше число: "))
    except ValueError:
        print("Пожалуйста, введите целое число.")
        continue
    
    # Проверка условий игры
    if user_guess == secret:
        print("Вы победили!")
        break  # Выходим из цикла, так как игрок угадал
    elif user_guess < secret:
        print("Загаданное число больше")
    else:
        print("Загаданное число меньше")
        
    # Уменьшаем счётчик попыток на 1
    attempts -= 1

# Если цикл закончился, а break не сработал — попытки исчерпаны
if attempts == 0:
    print(f"\nВы проиграли. Загаданное число: {secret}")
